# ===========================================================================
#  ANALYSIS CARD
# ===========================================================================
#  Edit the settings in this file to choose which collisions to keep and
#  which detector cuts to apply, then run
#
#      python analysis.py
#
#  You never need to edit any Python file to change a cut. This card is the
#  only input analysis.py takes.
#
#  An event is kept only if it is one of the collision types chosen under
#  FINAL STATE below, AND every particle in it passes every cut that is
#  switched on. If even one particle fails, the whole event is dropped.
#
#  Rules for this file:
#    - one setting per line, written as   name = value
#    - anything after a # is a comment and is ignored
#    - blank lines are ignored
#    - write  none  to switch a cut off entirely
#    - order does not matter
# ===========================================================================


# ---------------------------------------------------------------------------
#  FINAL STATE (COLLISION TYPE)
# ---------------------------------------------------------------------------
#  Choose which collision types to keep, by their ID:
#
#      ID   collision type
#      --   --------------------------
#       1   p p  ->  p p
#       2   p p  ->  gamma p p
#       3   p p  ->  n nbar p p
#       4   p p  ->  e+ e- p p
#       5   p p  ->  mu- mu+ p p
#       6   p p  ->  gamma gamma p p
#
#  Write  all  to keep every collision type.
#  Write one ID to keep only that type, e.g.   final_state = 4
#  Write several IDs separated by commas to keep more than one type, e.g.
#  final_state = 2, 6   keeps the two processes that contain photons.
#
#  The cuts below still apply on top of this choice.

final_state = all


# ---------------------------------------------------------------------------
#  ENERGY CUT
# ---------------------------------------------------------------------------
#  Drop any particle carrying less energy than this. Very soft particles are
#  lost in detector noise in a real experiment.
#  Units: TeV.  Range: 0 to 13.6.  Set to 0 (or none) to switch off.

min_energy = 0.0


# ---------------------------------------------------------------------------
#  ANGLE CUT
# ---------------------------------------------------------------------------
#  Keep a particle only if its polar angle theta falls inside this window.
#  theta is measured away from the beam line: 0 deg is straight down the beam,
#  90 deg is straight out the side, 180 deg is straight back.
#
#  Angles near 0 and 180 point into the beam pipe, where a real detector has no
#  coverage, so this is the beam-pipe cut.
#  Units: degrees.  Full range is 0 to 180, which switches the cut off.

theta_min_deg = 0.0
theta_max_deg = 180.0


# ---------------------------------------------------------------------------
#  PSEUDORAPIDITY CUT
# ---------------------------------------------------------------------------
#  An alternative way of writing the angle cut, in the units particle
#  physicists normally use:  eta = -ln(tan(theta/2)).
#
#  Keep a particle only if |eta| is below this value. The standard ATLAS/CMS
#  inner-detector acceptance is 2.5, which is the same as 9.385 deg < theta <
#  170.615 deg.
#
#  This overlaps with the theta cut above. Using both is fine; a particle has
#  to satisfy every cut that is switched on.
#  Set to  none  to switch off.

pseudo_rapidity_max = 2.5


# ---------------------------------------------------------------------------
#  TRANSVERSE MOMENTUM CUT
# ---------------------------------------------------------------------------
#  Keep a particle only if its momentum across the beam is at least this large.
#      pt = sqrt(px^2 + py^2) = |p| * sin(theta)
#
#  A detector measures pt rather than total energy, and a very energetic
#  particle skimming along the beam is useless because its sin(theta) is tiny.
#  A typical LHC value is 0.1 TeV (100 GeV).
#  Units: TeV.  Set to 0 (or none) to switch off.

transverse_momentum_min = 0.1


# ---------------------------------------------------------------------------
#  OUTPUT
# ---------------------------------------------------------------------------
#  yes = also write the surviving events to
#            analysis_tools/analysis_data/events_analysed.txt
#        in the same format as events.txt, so the graphing scripts can read it.
#  no  = just print the report and write nothing.

write_filtered_file = yes
